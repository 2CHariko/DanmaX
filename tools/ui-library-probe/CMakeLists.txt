cmake_minimum_required(VERSION 3.24)
project(UiLibraryProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Quick Test)
qt_standard_project_setup()
qt_add_executable(ui_library_probe main.cpp)
target_link_libraries(ui_library_probe PRIVATE Qt6::Quick Qt6::Test)
option(PROBE_SLIM_FLUENT "Build only the evaluated Fluent controls" OFF)
if(PROBE_SLIM_FLUENT)
    include(slim-fluent.cmake)
else()
set(FLUENTUI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FLUENTUI_BUILD_STATIC_LIB OFF CACHE BOOL "" FORCE)
set(FLUENTUI_QML_PLUGIN_DIRECTORY "${CMAKE_BINARY_DIR}/imports/FluentUI" CACHE PATH "" FORCE)
add_subdirectory("${FLUENT_SOURCE}" fluent)
endif()
