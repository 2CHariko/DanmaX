if(NOT WIN32 OR NOT MSVC OR DANMAKU_STATIC_QT)
    message(FATAL_ERROR "The Acrylic probe requires dynamic Windows x64 Qt / MSVC.")
endif()
find_package(Qt6 6.11.0 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test CorePrivate GuiPrivate QuickPrivate
    PATHS "${PROJECT_SOURCE_DIR}/.tools/qt/6.11.0/msvc2022_64/lib/cmake/Qt6" NO_DEFAULT_PATH)
qt_standard_project_setup(REQUIRES 6.11)
if(NOT TARGET QWindowKit::Quick)
    if(NOT EXISTS "${PROJECT_SOURCE_DIR}/.deps/qwindowkit/CMakeLists.txt")
        message(FATAL_ERROR "Prepare the locked project-local QWindowKit before running this probe.")
    endif()
    set(QWINDOWKIT_BUILD_WIDGETS OFF CACHE BOOL "" FORCE)
    set(QWINDOWKIT_BUILD_QUICK ON CACHE BOOL "" FORCE)
    set(QWINDOWKIT_BUILD_STATIC ON CACHE BOOL "" FORCE)
    set(QWINDOWKIT_INSTALL OFF CACHE BOOL "" FORCE)
    set(QWINDOWKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PROJECT_SOURCE_DIR}/.deps/qwindowkit" qwindowkit EXCLUDE_FROM_ALL)
endif()
qt_add_executable(acrylic_popup_probe WIN32 main.cpp BackdropBridge.h)
qt_add_resources(acrylic_popup_probe probe_ui PREFIX "/probe"
    FILES Main.qml ProbeComboBox.qml ProbeSurface.qml)
target_compile_definitions(acrylic_popup_probe PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(acrylic_popup_probe PRIVATE /utf-8 /EHsc)
target_link_libraries(acrylic_popup_probe PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Test QWindowKit::Quick dwmapi windowsapp)
