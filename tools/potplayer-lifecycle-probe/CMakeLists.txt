cmake_minimum_required(VERSION 3.31)
project(PotPlayerLifecycleProbe LANGUAGES CXX)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(REAL_PATH "${CMAKE_BINARY_DIR}" BUILD_ROOT)
string(FIND "${BUILD_ROOT}/" "${PROJECT_ROOT}/out/build/" BUILD_PREFIX)
if(NOT BUILD_PREFIX EQUAL 0)
    message(FATAL_ERROR "Build the probe under project out/build.")
endif()
if(MSVC AND DEFINED DANMAKU_MSVC_INCLUDES_PREFIX)
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "${DANMAKU_MSVC_INCLUDES_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
file(READ "${PROJECT_ROOT}/toolchain/dependencies.lock.json" TOOL_LOCK)
string(JSON QT_VERSION GET "${TOOL_LOCK}" qtVersion)
find_package(Qt6 ${QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2
    PATHS "${PROJECT_ROOT}/.tools/qt/${QT_VERSION}/msvc2022_64/lib/cmake/Qt6" NO_DEFAULT_PATH)
qt_standard_project_setup(REQUIRES 6.11)
qt_add_executable(potplayer_lifecycle_probe Probe.cpp)
qt_add_resources(potplayer_lifecycle_probe ui PREFIX "/" FILES Probe.qml)
target_link_libraries(potplayer_lifecycle_probe PRIVATE Qt6::Quick Qt6::QuickControls2 windowsapp user32)
include("${PROJECT_ROOT}/cmake/CompilerOptions.cmake")
danmaku_compile_options(potplayer_lifecycle_probe)
